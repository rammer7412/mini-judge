<body>
  <h1>Mini Judge</h1>
  <p class="muted">문제 선택 → 코드 제출 → 결과 확인</p>

  <div class="row">
    <label>Problem:</label>
    <select id="problemSelect"></select>
    <button id="refreshBtn">새로고침</button>
  </div>

  <h3>Code (Python3)</h3>
  <textarea id="codeArea">print("Hello")</textarea>

  <div class="row" style="margin-top:12px;">
    <button id="submitBtn">Submit</button>
    <span id="statusText" class="muted"></span>
  </div>

  <h3>Result</h3>
  <pre id="resultBox">(no result)</pre>

<script>
  const problemSelect = document.getElementById('problemSelect');
  const refreshBtn = document.getElementById('refreshBtn');
  const submitBtn = document.getElementById('submitBtn');
  const codeArea = document.getElementById('codeArea');
  const statusText = document.getElementById('statusText');
  const resultBox = document.getElementById('resultBox');

  async function loadProblems() {
    statusText.textContent = 'Loading problems...';
    const res = await fetch('/problems');
    const data = await res.json();
    problemSelect.innerHTML = '';
    (data.problems || []).forEach(id => {
      const opt = document.createElement('option');
      opt.value = id;
      opt.textContent = id;
      problemSelect.appendChild(opt);
    });
    statusText.textContent = data.problems?.length ? 'Ready' : 'No problems found (check /data/problems/<id>/tests)';
  }

  async function pollResult(sid) {
    while (true) {
      const res = await fetch(`/submissions/${sid}`);
      const data = await res.json();
      // status: QUEUED/RUNNING/DONE 같은 걸 worker가 넣어준다고 가정
      statusText.textContent = `Status: ${data.status || ''}`;
      resultBox.textContent = JSON.stringify(data, null, 2);
      if (data.status && data.status !== 'QUEUED' && data.status !== 'RUNNING') break;
      await new Promise(r => setTimeout(r, 700));
    }
  }

  async function submit() {
    const pid = problemSelect.value;
    if (!pid) {
      alert('문제가 없습니다. 서버의 /data/problems/<id>/tests 를 확인하세요.');
      return;
    }
    submitBtn.disabled = true;
    statusText.textContent = 'Submitting...';
    resultBox.textContent = '(submitting...)';

    const res = await fetch(`/problems/${pid}/submit`, {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({ code: codeArea.value })
    });

    if (!res.ok) {
      const t = await res.text();
      submitBtn.disabled = false;
      statusText.textContent = 'Submit failed';
      resultBox.textContent = t;
      return;
    }

    const data = await res.json();
    const sid = data.submission_id;
    statusText.textContent = `Submitted: ${sid}`;
    await pollResult(sid);

    submitBtn.disabled = false;
  }

  refreshBtn.addEventListener('click', loadProblems);
  submitBtn.addEventListener('click', submit);

  loadProblems();
</script>
